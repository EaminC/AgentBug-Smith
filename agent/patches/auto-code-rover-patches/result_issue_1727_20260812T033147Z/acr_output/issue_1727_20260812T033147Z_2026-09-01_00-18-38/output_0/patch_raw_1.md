class BudgetControlMiddleware(MiddlewareBase):

        self,
        max_tokens: int,
        hint_message: str = (
            "You have reached the maximum token budget. "
            "Please wrap up immediately and provide a final "
            "concluding response without invoking any tools."
        ),
    ) -> None:
        self.max_tokens = max_tokens
        self.hint_message = hint_message
        self._accumulated_tokens = 0

    async def on_reasoning(
        self,
        agent,
        input_kwargs: dict,
        next_handler: Callable[..., AsyncGenerator],
    ) -> AsyncGenerator:
        # Before reasoning step, accumulate token usage from the last reply message(s)
        # The token usage info is on Msg. We assume input_kwargs may contain context or tool_choice
        token_usage = 0
        # Try to get the last reply Msg from agent state or input kwargs if available
        # This depends on how the agent and reasoning pipeline supplies messages
        
        # Since we lack exact context, accumulate from agent.state or input_kwargs if possible
        # Here using getattr and fallback to 0 to be safe

        # For demonstration, check if agent.state has a latest reply message with usage info
        usage = 0
        messages = getattr(agent.state, "messages", None)
        if messages:
            # accumulate usage input + output tokens from assistant messages
            # Only from assistant messages to count tokens generated by the agent
            for msg in messages:
                if isinstance(msg, Msg) and msg.role == "assistant" and msg.usage:
                    usage += (msg.usage.input_tokens or 0) + (msg.usage.output_tokens or 0)

        # Update internal accumulated tokens
        self._accumulated_tokens = usage

        if self._accumulated_tokens >= self.max_tokens:
            # Yield a HintBlock event to signal agent to stop reasoning and wrap up

            # Yield a Msg containing this hint block

                name="BudgetControlMiddleware",
                role="system",
                content=[hint_block],
                metadata={},
                usage=None,
            )
            yield hint_msg
            # Do not proceed with further reasoning (stop here)
            return

        # Otherwise, continue normal reasoning by calling next_handler
        async for event in next_handler():
            yield event
